local noFogEnabled = true
local fogOriginal = { fogStart = Lighting.FogStart, fogEnd = Lighting.FogEnd }
local fogHidden = {}
local FOG_OBJECT_NAMES = { FantasySky = true, LightingLayers = true, Sky = true, SeaTerrorCC = true }

local function hideFogObject(obj)
	if fogHidden[obj] then return end
	fogHidden[obj] = obj.Parent
	obj.Parent = nil
end

local function applyNoFog()
	Lighting.FogEnd = math.huge
	for _, obj in ipairs(Lighting:GetChildren()) do
		if FOG_OBJECT_NAMES[obj.Name] or obj:IsA("Atmosphere") then
			hideFogObject(obj)
		end
	end
end

local function setNoFog(on)
	noFogEnabled = on
	if on then
		applyNoFog()
	else
		Lighting.FogStart = fogOriginal.fogStart
		Lighting.FogEnd = fogOriginal.fogEnd
		for obj, parent in pairs(fogHidden) do
			fogHidden[obj] = nil
			pcall(function() obj.Parent = parent or Lighting end)
		end
	end
end

Lighting.ChildAdded:Connect(function(obj)
	if noFogEnabled and (FOG_OBJECT_NAMES[obj.Name] or obj:IsA("Atmosphere")) then
		hideFogObject(obj)
	end
end)

local getupval = (debug and debug.getupvalue) or getupvalue
local setupval = (debug and debug.setupvalue) or setupvalue
local getupvals = (debug and debug.getupvalues) or getupvalues
local gunShootFunction = nil
local gunIdx = { v26 = 12, v22 = 13, v25 = 14, v21 = 15, v23 = 16, v24 = 17, v27 = 18 }

local function initGun()
	if gunShootFunction or not getupval then return end
	local controllers = ReplicatedStorage:FindFirstChild("Controllers")
	if not controllers and lp:FindFirstChild("PlayerScripts") then
		controllers = lp.PlayerScripts:FindFirstChild("Controllers")
	end
	local combat = controllers and controllers:FindFirstChild("CombatController")
	if not combat then combat = ReplicatedStorage:FindFirstChild("CombatController", true) end
	if not combat then return end
	local ok2, result = pcall(require, combat)
	if ok2 and type(result) == "table" and result.Attack then
		gunShootFunction = getupval(result.Attack, 9)
	end
end

local function nextValidator()
	if not gunShootFunction then initGun() end
	if not gunShootFunction or not getupvals or not getupval or not setupval then return 0, 0 end
	local upvals = getupvals(gunShootFunction)
	if not upvals then return 0, 0 end
	if upvals[gunIdx.v21] ~= 727595 then
		for i, v in pairs(upvals) do
			if v == 727595 then
				local offset = i - 15
				gunIdx.v21 = i
				gunIdx.v22 = 13 + offset
				gunIdx.v23 = 16 + offset
				gunIdx.v24 = 17 + offset
				gunIdx.v26 = 12 + offset
				gunIdx.v25 = 14 + offset
				gunIdx.v27 = 18 + offset
				break
			end
		end
	end
	local v1 = getupval(gunShootFunction, gunIdx.v21)
	local v2 = getupval(gunShootFunction, gunIdx.v22)
	local v3 = getupval(gunShootFunction, gunIdx.v23)
	local v4 = getupval(gunShootFunction, gunIdx.v24)
	local v5 = getupval(gunShootFunction, gunIdx.v25)
	local v6 = getupval(gunShootFunction, gunIdx.v26)
	local v7 = getupval(gunShootFunction, gunIdx.v27)
	if not (v1 and v2 and v3 and v4 and v5 and v6 and v7) then return 0, 0 end
	local v8 = v6 * v2
	local v9 = (v5 * v2 + v6 * v1) % v3
	v9 = (v9 * v3 + v8) % v4
	v5 = math.floor(v9 / v3)
	v6 = v9 - v5 * v3
	v7 = v7 + 1
	setupval(gunShootFunction, gunIdx.v25, v5)
	setupval(gunShootFunction, gunIdx.v26, v6)
	setupval(gunShootFunction, gunIdx.v27, v7)
	return math.floor(v9 / v4 * 16777215), v7
end

local function readNumberValue(inst)
	if not inst then return nil end
	local ok, val = pcall(function() return inst.Value end)
	if not ok then return nil end
	return tonumber(val)
end

local function getFragmentsNow()
	-- Lectura directa y robusta (tiempo real)
	local candidates = {}
	pcall(function()
		local data = lp:FindFirstChild("Data")
		if data then
			candidates[#candidates + 1] = data:FindFirstChild("Fragments")
			candidates[#candidates + 1] = data:FindFirstChild("Fragment")
		end
	end)
	pcall(function()
		if fragVal then candidates[#candidates + 1] = fragVal end
	end)
	pcall(function()
		local ls = lp:FindFirstChild("leaderstats")
		if ls then
			candidates[#candidates + 1] = ls:FindFirstChild("Fragments")
			candidates[#candidates + 1] = ls:FindFirstChild("Fragment")
		end
	end)
	pcall(function()
		local pd = lp:FindFirstChild("PlayerData") or lp:FindFirstChild("Stats")
		if pd then
			candidates[#candidates + 1] = pd:FindFirstChild("Fragments", true)
		end
	end)
	for _, v in ipairs(candidates) do
		local n = readNumberValue(v)
		if n ~= nil then return n end
	end
	-- último intento: búsqueda profunda
	local found = 0
	pcall(function()
		for _, inst in ipairs(lp:GetDescendants()) do
			if (inst:IsA("NumberValue") or inst:IsA("IntValue")) and inst.Name:lower():find("fragment") then
				found = tonumber(inst.Value) or found
				break
			end
		end
	end)
	return found
end

local function getEyesNow()
	local n = 0
	-- 1) función del hub si existe
	pcall(function()
		if tyrantEyes then
			local v = tyrantEyes()
			if type(v) == "number" then n = v end
		end
	end)
	if n > 0 then return n end
	-- 2) lectura directa del mapa (mismo criterio que tyrantEyes)
	pcall(function()
		local map = Workspace:FindFirstChild("Map")
		if not map then return end
		local tiki = map:FindFirstChild("TikiOutpost")
		if not tiki then return end
		local island = tiki:FindFirstChild("IslandModel") or tiki
		local eyes = {
			island:FindFirstChild("Eye1"),
			island:FindFirstChild("Eye2"),
		}
		local chunks = island:FindFirstChild("Chunks")
		local e = chunks and chunks:FindFirstChild("E")
		if e then
			eyes[3] = e:FindFirstChild("Eye3")
			eyes[4] = e:FindFirstChild("Eye4")
		else
			eyes[3] = island:FindFirstChild("Eye3", true)
			eyes[4] = island:FindFirstChild("Eye4", true)
		end
		local c = 0
		for i = 1, 4 do
			if eyes[i] and eyes[i].Transparency == 0 then c = c + 1 end
		end
		n = c
	end)
	return n
end

local function formatNumWH(n)
	n = math.floor(tonumber(n) or 0)
	local s = tostring(n)
	while true do
		local k
		s, k = s:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
		if k == 0 then break end
	end
	return s
end

local sessionFragStart = nil
local sessionFragReady = false
local sessionMoneyStart = nil
local sessionMoneyReady = false

local function getMoneyNow()
	local candidates = {}
	local function add(inst)
		if inst then candidates[#candidates + 1] = inst end
	end
	pcall(function()
		local data = lp:FindFirstChild("Data")
		if data then
			add(data:FindFirstChild("Beli"))
			add(data:FindFirstChild("Money"))
			add(data:FindFirstChild("Cash"))
			add(data:FindFirstChild("beli"))
		end
	end)
	pcall(function()
		local ls = lp:FindFirstChild("leaderstats")
		if ls then
			add(ls:FindFirstChild("Beli"))
			add(ls:FindFirstChild("Money"))
			add(ls:FindFirstChild("Cash"))
		end
	end)
	pcall(function()
		if moneyVal then add(moneyVal) end
	end)
	pcall(function()
		local pd = lp:FindFirstChild("PlayerData") or lp:FindFirstChild("Stats")
		if pd then
			add(pd:FindFirstChild("Beli", true))
			add(pd:FindFirstChild("Money", true))
		end
	end)
	-- búsqueda profunda por si el path cambió
	pcall(function()
		for _, inst in ipairs(lp:GetDescendants()) do
			if (inst:IsA("NumberValue") or inst:IsA("IntValue")) then
				local n = inst.Name:lower()
				if n == "beli" or n == "money" or n == "cash" then
					add(inst)
					break
				end
			end
		end
	end)
	for _, v in ipairs(candidates) do
		local n = readNumberValue(v)
		if n ~= nil then return n end
	end
	return 0
end


local function ensureSessionFragStart(force)
	local now = getFragmentsNow()
	if force or not sessionFragReady then
		sessionFragStart = now
		sessionFragReady = true
	elseif sessionFragStart ~= nil and now < sessionFragStart then
		sessionFragStart = now
	end
	local mon = getMoneyNow()
	if force or not sessionMoneyReady then
		sessionMoneyStart = mon
		sessionMoneyReady = true
	elseif sessionMoneyStart ~= nil and mon < sessionMoneyStart then
		sessionMoneyStart = mon
	end
	return now
end

local function getMoneySessionInfo()
	ensureSessionFragStart(false)
	local now = getMoneyNow()
	local start = sessionMoneyStart or now
	local gained = math.max(0, (now or 0) - (start or 0))
	return now, start, gained
end

local function getFragSessionInfo()
	local now = ensureSessionFragStart(false)
	local start = sessionFragStart or now
	local gained = math.max(0, (now or 0) - (start or 0))
	return now, start, gained
end

local function fragSessionText()
	local now, start, gained = getFragSessionInfo()
	local mNow, mStart, mGain = getMoneySessionInfo()
	return "Fragmentos: **" .. formatNumWH(now) .. "** (+" .. formatNumWH(gained) .. " sesión)\n"
		.. "Money: **$" .. formatNumWH(mNow) .. "** (+$" .. formatNumWH(mGain) .. " sesión)"
end

local function SendWebhook(title, description, color)
	if not Config.WebhookEnabled then return end
	local url = tostring(Config.WebhookURL or "")
	if (url == "" or not url:find("discord.com/api/webhooks")) and getgenv then
		url = tostring(getgenv().AndromedaWebhookURL or "")
	end
	if url == "" or not url:find("discord.com/api/webhooks") then return end
	if not httpRequest then return end
	if webhookBusy then return end
	if tick() - lastWebhookAt < 1.2 then return end
	webhookBusy = true
	lastWebhookAt = tick()
	task.spawn(function()
		pcall(function()
			local nowFrag, startFrag, gainedFrag = getFragSessionInfo()
			local nowMoney, _, gainedMoney = getMoneySessionInfo()
			local eyesCount = getEyesNow()
			local body = {
				username = (Config.WebhookUsername and Config.WebhookUsername ~= "") and Config.WebhookUsername or "Andromeda",
				embeds = {{
					title = tostring(title or "Andromeda"),
					color = color or 0x8B5CF6,
					fields = {
						{ name = "Fragments", value = formatNumWH(nowFrag), inline = true },
						{ name = "Frags sesión", value = "+" .. formatNumWH(gainedFrag), inline = true },
						{ name = "Dinero", value = "$" .. formatNumWH(nowMoney), inline = true },
						{ name = "Dinero sesión", value = "+$" .. formatNumWH(gainedMoney), inline = true },
						{ name = "Eyes", value = tostring(eyesCount) .. "/4", inline = true },
						{ name = "Player", value = lp.DisplayName .. " (@" .. lp.Name .. ")", inline = true },
						{ name = "Status", value = tostring(statusText or "—"), inline = false },
					},
					footer = { text = "Andromeda Hub · Tyrant" },
					timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ"),
				}},
			}
			httpRequest({
				Url = url,
				Method = "POST",
				Headers = { ["Content-Type"] = "application/json" },
				Body = HttpService:JSONEncode(body),
			})
		end)
		task.wait(0.3)
		webhookBusy = false
	end)
end

local lastEyesWebhook = -1
local lastEyesWebhookAt = 0
local function watchEyesWebhook()
	if not Config.WebhookOnEyes then return end
	local eyes = getEyesNow()
	if eyes ~= lastEyesWebhook then
		-- Solo avisar cuando sube (o llega a 4), no spam al bajar/reset
		local shouldSend = false
		if lastEyesWebhook < 0 then
			lastEyesWebhook = eyes
			return
		end
		if eyes > lastEyesWebhook or eyes == 4 then
			shouldSend = true
		end
		if shouldSend and tick() - lastEyesWebhookAt > 3 then
			lastEyesWebhookAt = tick()
			local msg
			if eyes >= 4 then
				msg = "Eyes **" .. tostring(eyes) .. "/4** — listo para **vasijas** / summon."
			else
				msg = "Progreso de ojos: **" .. tostring(eyes) .. "/4**"
			end
			SendWebhook("Eyes Tyrant", msg .. "\n" .. fragSessionText(), eyes >= 4 and 0xFBBF24 or 0xA78BFA)
		end
		lastEyesWebhook = eyes
	end
end

local function loadConfig()
	pcall(function()
		if isfile and isfile(CONFIG_FILE) and readfile then
			local data = HttpService:JSONDecode(readfile(CONFIG_FILE))
			if typeof(data) == "table" then
				for k, v in pairs(data) do
					if Config[k] ~= nil then Config[k] = v end
				end
			end
		end
	end)
	-- Fallback getgenv (misma sesión / executor)
	pcall(function()
		if getgenv then
			if type(getgenv().AndromedaWebhookURL) == "string" and getgenv().AndromedaWebhookURL ~= "" then
				Config.WebhookURL = getgenv().AndromedaWebhookURL
			end
			if getgenv().AndromedaConfig and type(getgenv().AndromedaConfig) == "table" then
				for k, v in pairs(getgenv().AndromedaConfig) do
					if Config[k] ~= nil then Config[k] = v end
				end
			end
		end
	end)
end

local function saveConfig()
	pcall(function()
		if writefile then
			writefile(CONFIG_FILE, HttpService:JSONEncode(Config))
		end
	end)
	pcall(function()
		if getgenv then
			getgenv().AndromedaWebhookURL = Config.WebhookURL
			getgenv().AndromedaConfig = Config
		end
	end)
end

					end
				end
			end
		end
	end
end)

local function safeDest(root)
	local y = Config.FarmDistance or 30
	local base
	if typeof(root) == "CFrame" then
		base = root
	elseif typeof(root) == "Instance" and root:IsA("BasePart") then
		base = root.CFrame
	elseif typeof(root) == "Instance" and root:IsA("Model") then
		local r = root:FindFirstChild("HumanoidRootPart") or root.PrimaryPart
		if not r then return nil end
		base = r.CFrame
	else
		return nil
	end
	local dest = (Config.PosMethod == "Below")
		and (base * CFrame.new(0, -math.abs(y), 0))
		or (base * CFrame.new(0, math.abs(y), 0))
	local pos = dest.Position
	if pos.Y < WATER_MIN_Y + math.abs(y) then
		dest = CFrame.new(pos.X, math.max(pos.Y, WATER_MIN_Y + math.abs(y)), pos.Z) * (dest - dest.Position)
	end
	return dest
end

local activeMoveTween = nil
local activeMoveDest = nil
local moveToken = 0

local function stopTween()
	moveToken = moveToken + 1
	if activeMoveTween then
		pcall(function() activeMoveTween:Cancel() end)
		activeMoveTween = nil
	end
	activeMoveDest = nil
	local hrp = getHRP()
	if hrp then
		local old = hrp:FindFirstChild("PartVelocuty")
		if old then pcall(function() old:Destroy() end) end
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
	end
end

local function clampDestToTiki(destCF)
	if typeof(destCF) ~= "CFrame" then return TIKI_FALLBACK end
	local p = destCF.Position
	if (p - TIKI_FALLBACK.Position).Magnitude > (TIKI_ZONE_RADIUS or 1200) then
		return TIKI_FALLBACK
	end
	if p.Magnitude < 500 then
		return TIKI_FALLBACK
	end
	return destCF
end

local function tweenTo(dest, speedMul)
	if antiBountyResetting or zHoldActive or not dest then return end
	local hrp = getHRP()
	local hum = getHum()
	if not hrp or not hum or hum.Health <= 0 then return end

	if typeof(dest) == "Vector3" then
		dest = CFrame.new(dest)
	elseif typeof(dest) == "Instance" and dest:IsA("BasePart") then
		dest = dest.CFrame
	elseif typeof(dest) == "Instance" and dest:IsA("Model") then
		local r = dest:FindFirstChild("HumanoidRootPart") or dest.PrimaryPart
		if not r then return end
		dest = r.CFrame
	elseif typeof(dest) ~= "CFrame" then
		return
	end

	if Config.AutoTyrant then
		dest = clampDestToTiki(dest)
	end

	if activeMoveDest and (dest.Position - activeMoveDest.Position).Magnitude < 5 and activeMoveTween then
		return
	end

	local dist = (dest.Position - hrp.Position).Magnitude
	if dist < 2 then
		hrp.CFrame = dest
		return
	end

	-- Cancelar tween anterior (sin BodyVelocity)
	if activeMoveTween then
		pcall(function() activeMoveTween:Cancel() end)
		activeMoveTween = nil
	end

	hrp.AssemblyLinearVelocity = Vector3.zero
	hrp.AssemblyAngularVelocity = Vector3.zero

	local speed = math.max((tonumber(Config.TweenSpeed) or 220) * (speedMul or 1), 50)
	local t = math.clamp(dist / speed, 0.08, 6)
	local token = moveToken
	local tw = TweenService:Create(hrp, TweenInfo.new(t, Enum.EasingStyle.Linear), { CFrame = dest })
	activeMoveTween = tw
	activeMoveDest = dest
	tw:Play()
	tw.Completed:Connect(function()
		if activeMoveTween == tw then
			activeMoveTween = nil
			activeMoveDest = nil
		end
	end)
end

local function holdAbove(root)
	if not root then return end
	local pos
	if typeof(root) == "Instance" and root:IsA("BasePart") then
		pos = root.Position
	elseif typeof(root) == "Instance" and root:IsA("Model") then
		local r = root:FindFirstChild("HumanoidRootPart") or root.PrimaryPart
		pos = r and r.Position
	elseif typeof(root) == "CFrame" then
		pos = root.Position
	end
	if Config.AutoTyrant and pos and (pos - TIKI_FALLBACK.Position).Magnitude > (TIKI_ZONE_RADIUS or 1200) then
		statusText = "Target fuera de Tiki — ignoro"
		tweenTo(TIKI_FALLBACK)
		return
	end
	local dest = safeDest(root)
	if dest then tweenTo(dest) end
end

local function gunAttack(target)
	local tool = findToolByTip("Gun")
	if not tool then return false end
	local hum = getHum()
	if hum and tool.Parent == lp.Backpack then
		pcall(function() hum:EquipTool(tool) end)
	end
	local part = target:FindFirstChild("Head") or target:FindFirstChild("HumanoidRootPart")
	if not part then return false end
	local now = tick()
	if now - lastGunShot < 0.05 then return false end
	lastGunShot = now

		pcall(function() Validator2:FireServer(code, count) end)

		tool:SetAttribute("LocalTotalShots", (tool:GetAttribute("LocalTotalShots") or 0) + 1)
	end)
	if ShootGunEvent then
		pcall(function() ShootGunEvent:FireServer(part.Position, { part }) end)
	end
	return true
end

local function canUseSkillZ(toolName)
	local ok2, ready = pcall(function()
		local skills = lp.PlayerGui:FindFirstChild("Main") and lp.PlayerGui.Main:FindFirstChild("Skills")
		if not skills then return true end
		local frame = skills:FindFirstChild(toolName)
		if not frame then return true end
		local z = frame:FindFirstChild("Z")
		if not z then return true end
		local cd = z:FindFirstChild("Cooldown")
		local title = z:FindFirstChild("Title")
		if title and title:IsA("TextLabel") then
			local c = title.TextColor3
			if c.R < 0.9 or c.G < 0.9 or c.B < 0.9 then return false end
		end
		if cd and cd:IsA("GuiObject") then
			local sx = cd.Size.X.Scale
			if sx > 0.05 and sx < 0.95 then return false end
		end
		return true
	end)
	return (not ok2) or ready
end

local function useFruitZ()
	-- Equip fruit (varios intentos)
	local fruit = nil
	for _ = 1, 8 do
		fruit = findToolByTip("Blox Fruit")
		if fruit then break end
		task.wait(0.15)
	end
	if not fruit then
		Notify(NOTIFY_NAME, "No Blox Fruit found for Z")
		return false
	end

	local hum = getHum()
	local char = getChar()
	if not hum or not char then return false end

	-- Equip
	for _ = 1, 10 do
		char = getChar()
		hum = getHum()
		if not char or not hum then return false end
		if fruit.Parent == char then break end
		if fruit.Parent == lp:FindFirstChild("Backpack") then
			pcall(function() hum:EquipTool(fruit) end)
		else
			fruit = findToolByTip("Blox Fruit")
			if not fruit then return false end
			pcall(function() hum:EquipTool(fruit) end)
		end
		task.wait(0.12)
	end

	-- Esperar a que la skill esté lista
	local ready = false
	for _ = 1, 15 do
		fruit = char:FindFirstChildOfClass("Tool")
		if fruit and fruit.ToolTip == "Blox Fruit" then
			if canUseSkillZ(fruit.Name) then
				ready = true
				break
			end
		else
			fruit = findToolByTip("Blox Fruit")
			if fruit then
				pcall(function()
					local h = getHum()
					if h then h:EquipTool(fruit) end
				end)
			end
		end
		task.wait(0.2)
	end

	-- Pulsar Z varias veces por si falla el input
	for _ = 1, 4 do
		pressKey("Z")
		task.wait(0.15)
	end
	task.wait(0.25)

	-- Volver al arma de farm
	pcall(equipWeapon)
	return true
end

-- Disparo real a una posición / part de vasija
local function shootSkullAt(tool, targetPos, targetPart)
	local char = getChar()
	if not char or not tool then return false end

	-- Equip forzado
	if tool.Parent ~= char then
		local hum = getHum()
		local bp = lp:FindFirstChild("Backpack")
		if hum and bp and tool.Parent == bp then
			pcall(function() hum:EquipTool(tool) end)
			for _ = 1, 12 do
				if tool.Parent == char then break end
				task.wait(0.04)
			end
		end
	end
	-- re-buscar por si el tool se recreó
	if tool.Parent ~= char then
		tool = findSkullTool()
		if not tool then return false end
		local hum = getHum()
		if hum and tool.Parent == lp:FindFirstChild("Backpack") then
			pcall(function() hum:EquipTool(tool) end)
			task.wait(0.1)
		end
	end
	if not tool or tool.Parent ~= char then return false end

	local my = getHRP()
	if not my then return false end

	local aimPos = targetPos
	if typeof(aimPos) ~= "Vector3" and targetPart and targetPart:IsA("BasePart") then
		aimPos = targetPart.Position
	end
	if typeof(aimPos) ~= "Vector3" then
		aimPos = my.Position + Vector3.new(0, -40, 0)
	end

	-- Throttle suave
	local now = tick()
	if now - lastSkullShot < 0.05 then
		task.wait(0.05)
	end
	lastSkullShot = tick()

	-- 1) Apuntar cámara a la vasija
	aimCameraAt(aimPos)

	-- 2) Reset overheat local (evita que deje de tirar)
	pcall(function()
		tool:SetAttribute("LocalOverheat", 0)
		tool:SetAttribute("LocalTotalShots", (tool:GetAttribute("LocalTotalShots") or 0) + 1)
	end)

	-- 3) Validator2 (necesario en muchos servers para que el tiro cuente)
	pcall(function()
		local code, count = nextValidator()
		if code ~= 0 and Validator2 then
			Validator2:FireServer(code, count)
		end
	end)

	-- 4) Remote del arma (TAP) si existe
	pcall(function()
		local remote = tool:FindFirstChild("RemoteEvent") or tool:FindFirstChild("Remote") or tool:FindFirstChildWhichIsA("RemoteEvent", true)
		if remote then
			remote:FireServer("TAP", aimPos)
			remote:FireServer(aimPos)
		end
	end)

	-- 5) ShootGunEvent global (el que usa el juego para guns)
	pcall(function()
		if ShootGunEvent then
			if targetPart and targetPart:IsA("BasePart") then
				ShootGunEvent:FireServer(aimPos, { targetPart })
				ShootGunEvent:FireServer(targetPart.Position, { targetPart })
			else
				ShootGunEvent:FireServer(aimPos, {})
			end
		end
	end)

	-- 6) Activate + clicks (backup local)
	pcall(function() tool:Activate() end)
	aimCameraAt(aimPos)
	doSkullM1()
	task.wait(0.03)
	aimCameraAt(aimPos)
	doSkullM1()

	return true
end

-- Alias por compatibilidad con el resto del script
local function shootSkullDown(tool, targetPos)
	return shootSkullAt(tool, targetPos, nil)
end

local fastAttackRemote, fastAttackId = nil, nil
local FAST_REMOTE_FOLDERS = { "Util", "Common", "Remotes", "Assets", "FX" }
local FAST_TARGET_FOLDERS = { "Enemies", "Characters" }
local function checkFastAttackChild(child)
	if child:IsA("RemoteEvent") and child:GetAttribute("Id") then
		fastAttackId = child:GetAttribute("Id")
		fastAttackRemote = child
	end
end
for _, folderName in ipairs(FAST_REMOTE_FOLDERS) do
	local folder = ReplicatedStorage:FindFirstChild(folderName)
	if folder then
		for _, child in ipairs(folder:GetChildren()) do checkFastAttackChild(child) end
		folder.ChildAdded:Connect(checkFastAttackChild)
	end
end
pcall(function() require(Net):RemoteEvent("RegisterHit", true) end)

local function doFastAttack()
	local char = getChar()
	if not char then return end
	local myHum = char:FindFirstChildOfClass("Humanoid")
	if not myHum or myHum.Health <= 0 then return end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	local tool = char:FindFirstChildOfClass("Tool")
	if not tool then return end
	local weaponType = tool:GetAttribute("WeaponType")
	if weaponType ~= "Melee" and weaponType ~= "Sword" then
		if tool.ToolTip ~= "Melee" and tool.ToolTip ~= "Sword" then return end
	end
	local range = Config.FastAttackRange or 5000
	local targets = {}
	for _, folderName in ipairs(FAST_TARGET_FOLDERS) do
		local folder = Workspace:FindFirstChild(folderName)
		if folder then
			for _, model in ipairs(folder:GetChildren()) do
				if model ~= char and model:IsA("Model") then
					local mhrp = model:FindFirstChild("HumanoidRootPart")
					local hum = model:FindFirstChildOfClass("Humanoid")
					if mhrp and hum and hum.Health > 0 and (mhrp.Position - hrp.Position).Magnitude <= range then
						for _, part in ipairs(model:GetChildren()) do
							if part:IsA("BasePart") then
								targets[#targets + 1] = { model, part }
							end
						end
					end
				end
			end
		end
	end
	if #targets == 0 then return end
	local head = nil
	for _, t in ipairs(targets) do
		local h = t[1]:FindFirstChild("Head")
		if h then head = h break end
	end
	if not head then return end
	pcall(function()
		local ra = Net:FindFirstChild("RE/RegisterAttack") or registerAttack
		if ra then ra:FireServer(ra) end
		local rh = Net:FindFirstChild("RE/RegisterHit") or registerHit
		if rh then
			local token = tostring(lp.UserId):sub(2, 4) .. tostring(coroutine.running()):sub(11, 15)
			rh:FireServer(rh, head, targets, {}, token)
		end
		local id = tonumber(fastAttackId)
		if fastAttackRemote and id then
			local remote = fastAttackRemote
			if cloneref then pcall(function() remote = cloneref(remote) end) end
			local seed = 0
			local seedObj = Net:FindFirstChild("seed")
			if seedObj then
				local ok2, result = pcall(function() return seedObj:InvokeServer() end)
				if ok2 and type(result) == "number" then seed = result end
			end
			local encoded = string.gsub("RE/RegisterHit", ".", function(c)
				return string.char(bit32.bxor(string.byte(c), math.floor(Workspace:GetServerTimeNow() / 10 % 10) + 1))
			end)
			remote:FireServer(remote, encoded, bit32.bxor(id + 909090, seed * 2), head, targets)
		end
	end)
end

	end
end

			if code ~= 0 and Validator2 then pcall(function() Validator2:FireServer(code, count) end) end
			local remote = tool:FindFirstChild("RemoteEvent")
			if remote then
				pcall(function() remote:FireServer("TAP", part.Position) end)
			elseif ShootGunEvent then
				pcall(function() ShootGunEvent:FireServer(part.Position, { part }) end)
			end
		end
		return
	end
	equipWeapon()
	local model = part:FindFirstAncestorOfClass("Model") or part
	sizepart(model)
	local hits = { { model, part } }
	pcall(function()
		if registerAttack then registerAttack:FireServer(0) end
		if registerHit then
			local first = table.remove(hits, 1)
			registerHit:FireServer(first[2], hits)
		end
	end)
	pcall(function()
		VirtualUser:CaptureController()
		VirtualUser:ClickButton1(Vector2.new(-1, 1))
	end)
end

local function isNearTiki(pos, maxDist)
	maxDist = maxDist or 900
	if typeof(pos) ~= "Vector3" then return false end
	-- Rechaza pivots basura (origen / otra isla / Hydra)
	if (pos - TIKI_FALLBACK.Position).Magnitude > maxDist then return false end
	if pos.Magnitude < 500 then return false end -- cerca de 0,0,0 = inválido
	return true
end

local function attackTree()
	local map = getMap()
	if not map then return false end
	local tiki = map:FindFirstChild("TikiOutpost")
	if not tiki then return false end
	local island = tiki:FindFirstChild("IslandModel")
	if not island then return false end
	local arena = island:FindFirstChild("EagleBossArena", true)
	if not arena then
		statusText = "Vase: no arena en Tiki"
		tweenTo(TIKI_FALLBACK)
		return false
	end
	local hrp = getHRP()
	if not hrp then return false end

	-- Elegir vasija SOLO si el pivot está cerca de Tiki (evita ir a Hydra)
	local bestTree, bestDist = nil, 1e9
	local function consider(child, requireAlive)
		if child.Name ~= "Tree" or not child:IsA("Model") then return end
		if requireAlive and child:GetAttribute("AlreadyDestroyedClient") then return end
		local okp, pivotCF = pcall(function() return child:GetPivot() end)
		if not okp or not pivotCF then return end
		local pivot = pivotCF.Position
		if not isNearTiki(pivot, 900) then return end
		-- Preferir la más cercana al centro de Tiki (no al jugador, por si estás lejos)
		local dist = (pivot - TIKI_FALLBACK.Position).Magnitude
		if dist < bestDist then
			bestDist = dist
			bestTree = child
		end
	end
	for _, child in ipairs(arena:GetChildren()) do
		consider(child, true)
	end
	if not bestTree then
		for _, child in ipairs(arena:GetChildren()) do
			consider(child, false)
		end
	end
	if not bestTree then
		statusText = "Vase: sin vasijas válidas en Tiki"
		tweenTo(TIKI_FALLBACK)
		return false
	end

	local pivotCF = bestTree:GetPivot()
	if not isNearTiki(pivotCF.Position, 900) then
		statusText = "Vase: pivot inválido, vuelvo a Tiki"
		tweenTo(TIKI_FALLBACK)
		return false
	end
	local targetPart = nil
	local group = bestTree:FindFirstChild("Group")
	if group then
		for _, part in ipairs(group:GetChildren()) do
			if part:IsA("BasePart") then targetPart = part break end
		end
	end
	if not targetPart then targetPart = bestTree.PrimaryPart end
	if not targetPart then targetPart = bestTree:FindFirstChildWhichIsA("BasePart", true) end
	local pivotPos = pivotCF.Position

	local useSkull = Config.ShootGunTyrant
	local hover = useSkull and 12 or 15
	local reach = useSkull and 40 or 75
	local dest = pivotCF * CFrame.new(0, hover, 0)
	if dest.Position.Y < WATER_MIN_Y + hover then
		dest = CFrame.new(dest.Position.X, WATER_MIN_Y + hover, dest.Position.Z)
	end
	-- Seguridad final: no tween fuera de zona Tiki
	if not isNearTiki(dest.Position, 1000) then
		statusText = "Vase: destino fuera de Tiki, cancelado"
		tweenTo(TIKI_FALLBACK)
		return false
	end
	tweenTo(dest)

	local tool = nil
	if useSkull then
		tool = loadAndEquipSkull()
		if not tool then
			statusText = "Vase: no Skull/Soul Guitar"
			return true
		end
	end

	local arrived = false
	local deadline = tick() + 10
	repeat
		task.wait(0.05)
		tweenTo(dest)
		local my = getHRP()
		if not my then break end
		if (pivotPos - my.Position).Magnitude <= reach then arrived = true break end
	until tick() >= deadline

	if not arrived then
		statusText = "Vase: out of range → Tiki"
		tweenTo(TIKI_FALLBACK)
		return true
	end

	if useSkull then
		statusText = "Vase: Skull Guitar M1 → vasija"
		local t0 = tick()
		local shots = 0
		local maxTime = 18
		local waitBetween = Config.MobileSafe and 0.28 or 0.12
		local maxShots = 40
		while tick() - t0 < maxTime and Config.AutoTyrant and not zHoldActive and not antiBountyResetting do
			if not bestTree or not bestTree.Parent then break end
			if bestTree:GetAttribute("AlreadyDestroyedClient") then break end
			if shots >= maxShots then break end

			-- mantenerse encima de la vasija
			tweenTo(dest)

			-- refrescar part por si el modelo cambia
			if not targetPart or not targetPart.Parent then
				local group = bestTree:FindFirstChild("Group")
				if group then
					for _, part in ipairs(group:GetChildren()) do
						if part:IsA("BasePart") then
							targetPart = part
							break
						end
					end
				end
				if not targetPart then
					targetPart = bestTree.PrimaryPart or bestTree:FindFirstChildWhichIsA("BasePart", true)
				end
			end
			local aim = (targetPart and targetPart.Position) or pivotPos

			tool = loadAndEquipSkull() or tool
			if not tool then
				statusText = "Vase: equipando Skull Guitar..."
				task.wait(0.25)
			else
				-- 3 disparos seguidos por ciclo
				for _ = 1, 3 do
					if shootSkullAt(tool, aim, targetPart) then
						shots = shots + 1
					end
					task.wait(0.08)
				end
				statusText = "Vase: M1 x" .. tostring(shots)
			end
			task.wait(waitBetween)
		end
		return true
	end

	statusText = "Vase: " .. tostring(Config.WeaponType)
	attackVasePart(targetPart)
	return true
end

local function getBoss(name)
	local enemies = Workspace:FindFirstChild("Enemies")
	if enemies then
		for _, m in ipairs(enemies:GetChildren()) do
			if m.Name == name and isAlive(m) then
				local r = m:FindFirstChild("HumanoidRootPart") or m.PrimaryPart
				if r and (r.Position - TIKI_FALLBACK.Position).Magnitude <= TIKI_ZONE_RADIUS + 400 then
					return m
				end
			end
		end
	end
	for _, m in ipairs(Workspace:GetChildren()) do
		if m:IsA("Model") and m.Name == name and isAlive(m) then
			local r = m:FindFirstChild("HumanoidRootPart") or m.PrimaryPart
			if r and (r.Position - TIKI_FALLBACK.Position).Magnitude <= TIKI_ZONE_RADIUS + 400 then
				return m
			end
		end
	end
	return nil
end

local function findTyrantMobPrefer(name)
	local enemies = Workspace:FindFirstChild("Enemies")
	if not enemies then return nil end
	local best, bd = nil, 1e9
	for _, m in ipairs(enemies:GetChildren()) do
		if isAlive(m) and TYRANT_MOBS[m.Name] then
			if (not name) or m.Name == name then
				local r = m:FindFirstChild("HumanoidRootPart")
				if r and (r.Position - TIKI_FALLBACK.Position).Magnitude <= TIKI_ZONE_RADIUS then
					-- Preferir cerca del centro de Tiki
					local d = (r.Position - TIKI_FALLBACK.Position).Magnitude
					if d < bd then bd = d best = m end
				end
			end
		end
	end
	if not best and name then return findTyrantMobPrefer(nil) end
	return best
end

	if not Config.BringMobs or not anchorModel then return end
	local my = getHRP()
	if not my then return end
	if tick() - lastBring < 0.08 then return end
	lastBring = tick()
	pcall(function()

			sethiddenproperty(lp, "SimulationRadius", math.clamp((Config.BringRadius or 350) + 100, 100, 1000))
		end
		if setscriptable then
			pcall(function() setscriptable(lp, "SimulationRadius", true) end)
		end
	end)
	local aRoot = anchorModel:FindFirstChild("HumanoidRootPart") or anchorModel.PrimaryPart
	if not aRoot then return end
	local dest = aRoot.Position
	local enemies = Workspace:FindFirstChild("Enemies")

	local count = 0
	for _, m in ipairs(enemies:GetChildren()) do
		if m ~= anchorModel and m.Name == anchorModel.Name then
			local hum = m:FindFirstChildOfClass("Humanoid")
			local r = m:FindFirstChild("HumanoidRootPart")
			if hum and hum.Health > 0 and r and not false then
				if (r.Position - my.Position).Magnitude <= range then
					pcall(function()
						r.CFrame = CFrame.new(dest + Vector3.new(math.random(-2, 2), math.random(0, 2), math.random(-2, 2)))
						r.AssemblyLinearVelocity = Vector3.zero
						r.AssemblyAngularVelocity = Vector3.zero
						r.CanCollide = false
						hum.WalkSpeed = 0
						hum.JumpPower = 0
						hum.PlatformStand = true
					end)
					sizepart(m)
					count += 1
					if count >= 12 then break end
				end
			end
		end
	end
end




local lastTyrantDamageTick = tick()
local lastTyrantMobHp = {}
local stuckHopIndex = 0
-- Offsets SOLO relativos a Tiki (antes eran coords globales cerca de 0,0,0 y te mandaban lejos / hacia Hydra)
local STUCK_OFFSETS = {
	CFrame.new(40, 15, 0),
	CFrame.new(-40, 15, 0),
	CFrame.new(0, 15, 40),
	CFrame.new(0, 15, -40),
	CFrame.new(70, 20, 70),
	CFrame.new(-70, 20, -70),
	CFrame.new(0, 25, 0), -- centro Tiki

local function noteTyrantDamage(mob)
	if not mob then return end
	local hum = mob:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	local key = tostring(mob)
	local hp = hum.Health
	local prev = lastTyrantMobHp[key]
	if prev ~= nil and hp < prev - 0.5 then
		lastTyrantDamageTick = tick()
	end
	lastTyrantMobHp[key] = hp
	if hp <= 0 then
		lastTyrantDamageTick = tick()
		lastTyrantMobHp[key] = nil
	end
end

local function hopTyrantStuck()
	stuckHopIndex = (stuckHopIndex % #STUCK_OFFSETS) + 1
	local offset = STUCK_OFFSETS[stuckHopIndex] or CFrame.new()
	-- Siempre anclado a Tiki Outpost, nunca coords globales del mapa
	local dest = TIKI_FALLBACK * offset
	-- Evitar agua baja
	if dest.Position.Y < WATER_MIN_Y + 10 then
		dest = CFrame.new(dest.Position.X, WATER_MIN_Y + 20, dest.Position.Z)
	end
	print("[Andromeda] 30s sin daño → hop cerca de Tiki", stuckHopIndex)
	lockedTarget = nil
	lockedName = nil
	lastTyrantMobHp = {}
	lastTyrantDamageTick = tick()
	statusText = "Stuck 30s — hop Tiki"
	tweenTo(dest)
end

local function autoTyrantStep()
	if antiBountyResetting or zHoldActive then return end
	if not isSea3() then
		statusText = "No Sea 3"
		lockedTarget = nil
		return
	end

	-- Si te fuiste muy lejos de Tiki (ej. Hydra), volver siempre
	local hrpSafe = getHRP()
	if hrpSafe and (hrpSafe.Position - TIKI_FALLBACK.Position).Magnitude > 3500 then
		statusText = "Fuera de zona Tyrant — volviendo a Tiki"
		lockedTarget = nil
		lockedName = nil
		tweenTo(TIKI_FALLBACK)
		return
	end

	-- 30s sin daño en farm → hop ubicación cerca de Tiki
	if tick() - lastTyrantDamageTick >= 30 then
		hopTyrantStuck()
		return
	end

	local boss = getBoss("Tyrant of the Skies")
	if boss then
		lockedTarget = boss
		lockedName = boss.Name
		local root = boss:FindFirstChild("HumanoidRootPart") or boss.PrimaryPart
		if root then
			holdAbove(root)
			bringSameName(boss)
			attackTarget(boss)
			noteTyrantDamage(boss)
			statusText = "Killing Tyrant"
		end
		return
	end

	local eyes = tyrantEyes()
	if eyes >= 4 then
		lockedTarget = nil
		lockedName = nil
		-- Boss tiene prioridad: si existe, no seguir en vasijas
		if getBoss("Tyrant of the Skies") then
			-- cae al bloque de boss en el siguiente tick (arriba ya lo chequea)
			-- por si el chequeo de arriba falló por timing, matar aquí
			local boss = getBoss("Tyrant of the Skies")
			local root = boss and (boss:FindFirstChild("HumanoidRootPart") or boss.PrimaryPart)
			if root then
				holdAbove(root)
				bringSameName(boss)
				-- usar arma de farm, no guitarra
				attackTarget(boss)
				noteTyrantDamage(boss)
				statusText = "Killing Tyrant"
			end
			return
		end
		statusText = "Eyes 4/4, going to Vases"
		attackTree()
		return
	end

	if lockedTarget and isAlive(lockedTarget)  then
		local root = lockedTarget:FindFirstChild("HumanoidRootPart")
		if root then
			holdAbove(root)
			bringSameName(lockedTarget)
			attackTarget(lockedTarget)
			noteTyrantDamage(lockedTarget)
			statusText = "Farm " .. lockedTarget.Name .. " | Eyes " .. eyes .. "/4"
			return
		end
	end

	-- target inválido / invisible / stuck → soltar
	if lockedTarget then
		lockedTarget = nil
		lockedName = nil
	end

	lockedTarget = findTyrantMobPrefer(lockedName)
	if lockedTarget and isAlive(lockedTarget) then
		lockedName = lockedTarget.Name
		local root = lockedTarget:FindFirstChild("HumanoidRootPart")
		if root then
			holdAbove(root)
			bringSameName(lockedTarget)
			attackTarget(lockedTarget)
			noteTyrantDamage(lockedTarget)
			statusText = "Farm " .. lockedTarget.Name .. " | Eyes " .. eyes .. "/4"
			return
		end
	end
	lockedTarget = nil
	lockedName = nil

	lockedTarget = nil
	lockedName = nil
	local hrp = getHRP()
	if hrp and (hrp.Position - TIKI_FALLBACK.Position).Magnitude > 80 then
		tweenTo(TIKI_FALLBACK)
	end
	statusText = "Moving to Tiki | Eyes " .. eyes .. "/4"
end

local function findValue(root, names)
	if not root then return nil end
	for _, name in ipairs(names) do
		local v = root:FindFirstChild(name, true)

local function resolveStats()
	local roots = {
		lp:FindFirstChild("Data"),
		lp:FindFirstChild("PlayerData"),
		lp:FindFirstChild("leaderstats"),
		lp:FindFirstChild("Stats"),
		lp,
	}
	local money, frag
	for _, root in ipairs(roots) do
		if not money then
			money = findValue(root, { "Money", "Beli", "Cash", "beli", "money" })
		end
		if not frag then
			frag = findValue(root, { "Fragments", "Fragment", "fragments", "Frag" })
		end
		if money and frag then break end
	end
	return money, frag
end

local function formatNum(n)
	n = math.floor(tonumber(n) or 0)
	local neg = n < 0
	n = math.abs(n)
	local s = tostring(n)
	while true do
		local k
		s, k = s:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
		if k == 0 then break end
	end
	return (neg and "-" or "") .. s
end

local Fluent = { Options = {} }

local function RedzElements()
	local ok, els = pcall(function()
		return RedzWindow:GetElements()
	end)
	if ok then return els end
end
local ElementsRoot = RedzElements() or {}
local RootFrame = ElementsRoot.MainFrame or ElementsRoot.Components

local function Register(key, shim)
	if key then Fluent.Options[key] = shim end
	return shim
end

local function AddToggle(tabRaw, key, cfg)
	cfg = cfg or {}
	local default = cfg.Default == true
	local shim = { Value = default, Key = key }
	local building = true
	local raw = tabRaw:AddToggle({
		Name = cfg.Title or key or "-",
		Description = cfg.Description or "",
		Default = default,
		Callback = function(value)
			shim.Value = value == true
			if building then return end
			if cfg.Callback then
				pcall(cfg.Callback, shim.Value)
			end
		end,
	})
	building = false
	shim.SetValue = function(_, v)
		shim.Value = v == true
		pcall(function() raw:Set(shim.Value) end)
	end
	return Register(key, shim)
end

local function AddDropdown(tabRaw, key, cfg)
	cfg = cfg or {}
	local values = cfg.Values or {}
	local default = cfg.Default
	local shim = { Value = default, Key = key }
	local building = true
	local raw = tabRaw:AddDropdown({
		Name = cfg.Title or key or "-",
		Description = cfg.Description or "",
		Options = values,
		Default = default,
		Callback = function(value)
			if type(value) == "table" then value = value[1] end
			shim.Value = value
			if building then return end
			if cfg.Callback then pcall(cfg.Callback, value) end
		end,
	})
	building = false
	shim.SetValue = function(_, v)
		shim.Value = v
		pcall(function() raw:Set(v) end)
	end
	return Register(key, shim)
end

local function AddSlider(tabRaw, key, cfg)
	cfg = cfg or {}
	local default = tonumber(cfg.Default) or 0
	local shim = { Value = default, Key = key }
	local building = true
	local raw = tabRaw:AddSlider({
		Name = cfg.Title or key or "-",
		Description = cfg.Description or "",
		Min = cfg.Min or 0,
		Max = cfg.Max or 100,
		Default = default,
		Increment = cfg.Increment or 1,
		Callback = function(value)
			shim.Value = value
			if building then return end
			if cfg.Callback then pcall(cfg.Callback, value) end
		end,
	})
	building = false
	return Register(key, shim)
end

local function AddInput(tabRaw, key, cfg)
	cfg = cfg or {}
	local default = tostring(cfg.Default or "")
	local shim = { Value = default, Key = key }
	local building = true
	local raw = tabRaw:AddTextBox({
		Name = cfg.Title or key or "-",
		Description = cfg.Description or "",
		Default = default,
		Placeholder = cfg.Placeholder or "",
		Callback = function(value)
			shim.Value = tostring(value or "")
			if building then return end
			if cfg.Callback then pcall(cfg.Callback, shim.Value) end
		end,
	})
	building = false
	-- fallback APIs
	if not raw then
		pcall(function()
			raw = tabRaw:AddInput({
				Name = cfg.Title or key or "-",
				Default = default,
				Callback = function(value)
					shim.Value = tostring(value or "")
					if cfg.Callback then pcall(cfg.Callback, shim.Value) end
				end,
			})
		end)
	end
	return Register(key, shim)
end

local function AddButton(tabRaw, cfg)
	cfg = cfg or {}
	return tabRaw:AddButton({
		Name = cfg.Title or "Button",
		Description = cfg.Description or "",
		Callback = function()
			if cfg.Callback then pcall(cfg.Callback) end
		end,
	})
end

local function NewSection(tabRaw, name)
	local sec = tabRaw:AddSection(name)
	local api = {}
	api.AddToggle = function(_, key, c) return AddToggle(tabRaw, key, c) end
	api.AddDropdown = function(_, key, c) return AddDropdown(tabRaw, key, c) end
	api.AddSlider = function(_, key, c) return AddSlider(tabRaw, key, c) end
	api.AddInput = function(_, key, c) return AddInput(tabRaw, key, c) end
	api.AddButton = function(_, c) return AddButton(tabRaw, c) end
	return api
end

local function NewTab(cfg)
	local rawTab = RedzWindow:MakeTab({
		Title = cfg.Title or "Tab",
		Icon = cfg.Icon or "",
	})
	local shim = { Raw = rawTab }
	shim.AddSection = function(_, name) return NewSection(rawTab, name) end
	shim.AddToggle = function(_, key, c) return AddToggle(rawTab, key, c) end
	shim.AddDropdown = function(_, key, c) return AddDropdown(rawTab, key, c) end
	shim.AddSlider = function(_, key, c) return AddSlider(rawTab, key, c) end
	shim.AddInput = function(_, key, c) return AddInput(rawTab, key, c) end
	shim.AddButton = function(_, c) return AddButton(rawTab, c) end
	return shim
end

local Window = {
	Root = RootFrame,
	Minimized = false,
}
Window.AddTab = function(_, cfg) return NewTab(cfg) end
Window.SelectTab = function(_, index)
	pcall(function() RedzWindow:SelectTab(index) end)
end
Window.Notify = function(_, data)
	data = data or {}
	pcall(function()
		RedzWindow:Notify({
			Title = tostring(data.Title or HUB_NAME),
			Content = tostring(data.Content or ""),
			Time = data.Duration or 3,
		})
	end)
end
Fluent.Notify = function(_, data) Window:Notify(data) end

-- Override Notify to also use Fluent popup when available
local _OldNotify = Notify
Notify = function(title, content)
	_OldNotify(title, content)
	pcall(function()
		Fluent:Notify({ Title = title or NOTIFY_NAME, Content = content or "", Duration = 3 })
	end)
end
